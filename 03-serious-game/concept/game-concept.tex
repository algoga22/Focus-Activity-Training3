  \pgfmathsetmacro\lvl{#5}
  \ifdim\lvl pt>99pt
    \draw[alert, fill=alert!12, rounded corners=2mm, line width=.6mm] (#1,#2) rectangle ++(36,23);

    \draw[softdark, fill=white, rounded corners=2mm, line width=.6mm] (#1,#2) rectangle ++(36,23);
  \fi
  \node[text=navy, font=\Large] at ({#1+18},{#2+16.5}) {#3};
  \node[font=\headfont\scriptsize, text=navy] at ({#1+18},{#2+10.5}) {#4};

  \ifdim\lvl pt>0pt \fill[alert, rounded corners=.6mm] ({#1+5},{#2+2.5}) rectangle ++({26*#5/100},1.8);\fi}
