\newcommand\spot[4]{%
  \begin{tcolorbox}[enhanced, colback=white, colframe=soft, boxrule=.6pt, arc=2.5mm, left=2.5mm, right=2.5mm, top=2mm, bottom=2mm,
      equal height group=spot, width=\linewidth, before skip=0pt, after skip=0pt, before upper=\raggedright]
    \badge[7.5mm]{#1}{\small #2}\hspace{2mm}{\headfont\small\color{navy}#3}\par\vspace{1.2mm}
    {\footnotesize #4\par}
  \end{tcolorbox}}
% \step{<colour>}{<n>}{<icon>}{<name>}{<fire>}{<content>}
\newcommand\step[6]{%
  \begin{tcolorbox}[enhanced, colback=white, colframe=#1, boxrule=0pt, leftrule=2.2mm, arc=2.5mm, left=4mm, right=4mm, top=2.2mm, bottom=2.2mm,
      before skip=0pt, after skip=2.2mm, drop fuzzy shadow={black!25}]
    \begin{minipage}[t]{30mm}
      \kicker{#1}{STEP #2}\par\vspace{1mm}
      \badge[10mm]{#1}{\large #3}\par\vspace{1.2mm}
      {\headfont\large\color{navy}#4\par}\vspace{.6mm}
      {\scriptsize\color{muted}\faFire\ \emph{#5}\par}
    \end{minipage}\hfill
    \begin{minipage}[t]{\dimexpr\linewidth-34mm\relax}\vspace{0mm}#6\end{minipage}
  \end{tcolorbox}}

% ---------------------------------------------------------------- THE 4 STEPS
\kicker{alert}{2 \textbullet{} ONE RULE, FOUR STEPS}\par\vspace{2mm}
\small
\step{alert}{1}{\faHandPaper}{Stop}{Don't add fuel}{%
  \doit{Take your hands off the keyboard}
  \dontdo{Don't click, type, open or pay anything else}
  \dontdo{Don't try to fix it yourself}}
\step{navy}{2}{\faWifi}{Disconnect}{Close the door}{%
  \doit{Turn off the Wi-Fi or unplug the network cable}
  \doit{Leave the computer \textbf{on}: restarting or switching off can erase clues}
  \doit{Lost device or money sent? Nothing to unplug: \textbf{go straight to step 3}}}
\step{good}{3}{\faPhone}{Report}{Call the firefighters}{%
  \doit{\textbf{Call now}, even on Friday at 18:00. No answer? Call the backup.}
  \vspace{.6mm}
  {\mediumfont\renewcommand{\arraystretch}{1.9}\begin{tabular}{@{}l@{\ \ }l@{\qquad}l@{\ \ }l@{}}
    My contact: & \fillin{48mm} & Tel: & \fillin{32mm} \\
    Backup: & \fillin{48mm} & Tel: & \fillin{32mm} \\
    Our bank: & \fillin{48mm} & Tel: & \fillin{32mm} \\
  \end{tabular}\par}
  {\scriptsize\color{muted}Bank: for a payment to a fake account, call within minutes: it can sometimes be stopped.\par}}
\step{navydark}{4}{\faPen}{Write down}{Tell them what you saw}{%
  Four lines are enough. Keep the strange emails and files: they are clues.\par\vspace{1.4mm}
  \tick\ \textbf{When} it happened\qquad \tick\ \textbf{What} you saw\qquad \tick\ \textbf{What} you did and who you called\par\vspace{1.4mm}
  \tick\ A \textbf{photo of the screen} with your phone (or a screenshot)}
\vspace{2.5mm}
