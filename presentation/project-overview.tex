% ---------- The brief ----------
\begin{frame}{The brief and our audience}{Overview}
  \vspace{1mm}
  \begin{columns}[T,onlytextwidth]
    \column{.48\textwidth}
    \explain{What the bootcamp asked}{The Cybersecurity for NGOs Bootcamp (Digital Learning Hub, 13 and 16 November 2026) has three trainings. Ours is \textbf{Training 3: ``When something goes wrong: a simple incident response''}, \textbf{60 minutes}.
      We must deliver a training plan, a diagnostic assessment, a serious game, a learning-by-doing activity, a take-home sheet and the slides.
      Everything must work \textbf{online and onsite}, and the organisers ask us to \textbf{avoid account creation} and connection delays.}
    \explain{Our main design rule}{Simple, jargon-free, practical. Every technical word is explained or avoided, and every part of the hour leads to the same small set of actions.}
    \column{.48\textwidth}
    \explain{Who we train}{NGO staff and volunteers with \textbf{no technical background}: food banks, shelters, small associations. Typical situation:
      \bul{alert}{a small team and \textbf{no IT department};}
      \bul{alert}{sensitive data of beneficiaries, volunteers and donors;}
      \bul{alert}{shared computers, personal phones used for work;}
      \bul{alert}{coordinators often away; many volunteers part-time.}
      The problem we solve: when something goes wrong, \textbf{the first minutes decide the damage}, and most people either do nothing (embarrassment, ``it can wait'') or the wrong thing (restart, pay, delete).}
  \end{columns}
\end{frame}

% ---------- The idea ----------
\begin{frame}{One analogy, one rule}{Overview \textbullet{} the thread of the hour}
  \vspace{1mm}
  \begin{columns}[T,onlytextwidth]
    \column{.48\textwidth}
    \explain{\textcolor{alert}{\faFire}\ An incident is like a fire}{The sooner you spot it, the less damage it does. A bin on fire (minutes) is easy; the whole building (days) is not.
      The warning signs are the smoke, and \textbf{each person is the smoke alarm}: you don't need to be sure it is a fire to raise the alarm.
      This analogy appears in the slides (minutes 10--18), becomes the \textbf{rule of the game} (the fire really grows) and is the header of the cheat sheet.}
    \explain{\textcolor{alert}{\faTimes}\ Never \ \textcolor{good}{\faUserShield}\ No blame}{Never pay, delete, hide, restart or forward. Nobody gets blamed for reporting: the only mistake is to stay silent.}
    \column{.48\textwidth}
    \explain{One rule, four steps}{%
      \bul{alert}{\textbf{Stop} (don't add fuel): hands off; don't click, type, pay or try to fix it.}
      \bul{navy}{\textbf{Disconnect} (close the door): Wi-Fi off or cable out, \textbf{leave the computer on} (restarting erases clues). Lost device or money sent: nothing to unplug, go to step 3.}
      \bul{good}{\textbf{Report} (call the firefighters): call your contact \textbf{now}, even on Friday at 18:00; no answer, call the backup; money sent, call the bank.}
      \bul{navydark}{\textbf{Write down} (tell them what you saw): when, what you saw, what you did and who you called, a photo of the screen.}
      The same words are used in the slides, the quiz answers, the game phases, the case answer key and the cheat sheet.}
  \end{columns}
\end{frame}

% ---------- Quiz design ----------
\begin{frame}{Why the questions are built this way}{Challenge 2 \textbullet{} question design}
  \vspace{1mm}
  \begin{columns}[T,onlytextwidth]
    \column{.48\textwidth}
    \explain{So that a before / after comparison means something}{A first version had 3 options with an obvious right answer: people would have scored high before the training and no progress would show. Now:
      \bul{navy}{\textbf{4 options}: guessing gives only 25\,\%;}
      \bul{alert}{\textbf{the wrong options are real beliefs} (switch it off, run the antivirus, email the coordinator, ask a friend, guess the virus, forward to warn, reply to check), each corrected on a slide;}
      \bul{good}{\textbf{the right answer uses the words of the training}, so it is indisputable;}
      \bul{navydark}{\textbf{answers are hidden at the start}: the first round does not teach the second.}}
    \column{.48\textwidth}
    \explain{Against the Challenge 2 criteria}{%
      \bul{navy}{\textbf{Relevance (25):} 1 question per objective.}
      \bul{navy}{\textbf{Quality of questions (20):} real NGO situations, short, no tricks.}
      \bul{navy}{\textbf{Quality of answers (5):} indisputable right answer, plausible distractors.}
      \bul{alert}{\textbf{Measure progress (25):} the same quiz at minute 5 and minute 53, compared by the app.}
      \bul{navy}{\textbf{Tool choice (20):} 8 factual reasons (next slides).}
      \bul{navy}{\textbf{Coverage (5):} all five objectives, one each.}}
    \explain{Paper backup}{\file{answer-sheet.pdf}: same questions, a Start and an End column, two sheets per A4 page.}
  \end{columns}
\end{frame}

% ---------- Game concept ----------
\begin{frame}{Put out the fire: the idea}{Challenge 3 \textbullet{} serious game}
  \vspace{1mm}
  \begin{columns}[T,onlytextwidth]
    \column{.48\textwidth}
    \explain{The analogy becomes the rule}{In the slides we say ``the sooner you spot it, the less damage it does''. In the game it is literally true: \textbf{every second of hesitation makes the fire grow}, a right move is \textbf{water}, a classic mistake is a \textbf{trap card: petrol}. Players don't memorise the lesson, they feel it.}
    \explain{Structure}{Our NGO has \textbf{4 rooms = 4 rounds}, each with a real incident. Each round has \textbf{4 phases = the 4 steps}: Is it a fire? (recognise), Cut the fire (stop and disconnect), Call the firefighters (report), Fire report (write down). Every phase: 3 options, \textbf{1 right move and 2 trap cards}. 16 decisions in total.}
    \column{.48\textwidth}
    \explain{The 4 rooms}{%
      \bul{navy}{\textbf{Work phones:} Ana loses the work phone with the beneficiaries' WhatsApp groups on the bus.}
      \bul{navy}{\textbf{Social media:} the NGO's Facebook page posts an ad nobody wrote.}
      \bul{navy}{\textbf{Beneficiary data:} the volunteers' home addresses sent to the wrong ``Marc''.}
      \bul{navy}{\textbf{Office computers:} slow computer, files ending in ``.locked'', then a ransom message; \textbf{surprise:} a message from ``Anna (management)'' says ``pay now and don't tell anyone''.}}
    \explain{Mode}{One fire per room \textbf{for the whole team} plus \textbf{individual points} and a leaderboard. Every person's answers are recorded.}
  \end{columns}
\end{frame}

      & \textcolor{muted}{1 Is it a fire?} & \textcolor{muted}{2 Cut the fire} & \textcolor{muted}{3 Call the firefighters} & \textcolor{muted}{4 Fire report}\\\midrule
      \textbf{Work phones} & Yes: the beneficiaries' contacts are at risk & Nothing to unplug: report straight away & The backup contact, now & When and where it was last seen, what was on it\\

% ---------- Fire mechanics ----------
\begin{frame}{How the fire is calculated}{Challenge 3 \textbullet{} mechanics}
  \vspace{1mm}
  \begin{columns}[T,onlytextwidth]
    \column{.48\textwidth}
    \explain{During a phase (up to 25 s)}{The room's fire grows \textbf{1.6\,\% per second} on every screen and phone. The phase closes when everybody has answered (after at least 2 s) or after 25 s. All numbers on this slide are constants at the top of \file{quiz/game.js}.}
    \explain{When the phase closes}{With \(n\) players, \(g\) right answers and \(t\) trap cards:\\[1mm]
      \(\textit{fire} = \textit{start} + 1.6 \times \textit{seconds} - 40\,\tfrac{g}{n} + 30\,\tfrac{t}{n}\)\\[1mm]
      (between 0 and 100). No answer counts as nothing: only the time cost. The screen shows the three parts: waiting, water and petrol.}
    \column{.48\textwidth}
    \explain{End of a room}{%
      \bul{good}{\textbf{Saved:} under 50\,\% after phase 4.}
      \bul{alert}{\textbf{Damaged:} 50--99\,\%.}
      \bul{alert}{\textbf{Burnt down:} it reached 100\,\%. The fire \textbf{spreads}: the next room starts at 45\,\% instead of 20\,\%.}}
    \explain{Points}{A right move gives \textbf{100 points + up to 50 for speed} (50 if instant, 0 at 25 s). Trap card or no answer: 0. The first tap locks the answer. Leaderboard after each room; each phone shows the points and the rank.}
  \end{columns}
\end{frame}
